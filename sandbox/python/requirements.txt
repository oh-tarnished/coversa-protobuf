# Runtime dependencies for the generated Python, installed by CI before it
# imports every generated module.
#
# Not a package anyone installs: it exists so a schema producing code the
# descriptor pool rejects is caught here rather than by the first consumer.
# Compiling the files would not catch it -- the pool is only built at import.
#
# The floor tracks the BSR plugin, not taste. Generated Python opens with
# `ValidateProtobufRuntimeVersion`, which refuses a runtime older than the
# gencode that wrote it: protoc-gen-python currently emits gencode 7.36, so a
# 6.x runtime fails every module with a VersionError before a single
# descriptor is built. Raise this when the plugin moves.
protobuf>=7.36.1,<8

# google.api and google.type come from here. The generated tree contains its
# own copies -- the template generates with --include-imports, since generated
# Python does `from buf.validate import validate_pb2`, a path no PyPI package
# exposes -- and CI removes gen/python/google before importing, so the
# installed distribution is what resolves.
googleapis-common-protos>=1.75.3
